# Runtime dependencies for the AI Operations Toolkit backend.
#
# Each line installs one local package in editable mode. Every package declares
# its own third-party dependencies in its own pyproject.toml, so pip resolves
# the full tree from this one file:
#
#     python -m venv .venv
#     .venv/Scripts/python -m pip install -r requirements.txt      (Windows)
#     .venv/bin/python     -m pip install -r requirements.txt      (macOS/Linux)
#
# Order does not matter — pip resolves the graph.

-e ./packages/config
-e ./packages/utils
-e ./packages/ai
-e ./packages/database
-e ./packages/adapters

-e ./services/workflow-engine
-e ./services/analytics
-e ./services/document-processing

# Projects, in build order.
-e ./projects/ai-sop-generator
-e ./projects/ai-operations-dashboard
-e ./projects/ai-travel-operations

# The API app, which declares FastAPI and uvicorn. Installing it (rather than
# relying on the working directory) is what lets pytest and CI import `app`
# from anywhere in the repo.
-e ./apps/api
